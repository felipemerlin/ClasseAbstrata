namespace ClasseAbstrata
{
    class Program
    {
        static void Main(string[] args)
        {
            List<IValidavel> campos = new List<IValidavel>
        {
            new CampoTexto("Nome", "Felipe"),
            new CampoTexto("Sobrenome", "Merlin"),
            new CampoNumerico("Idade", "16")
        };

        foreach (var campo in campos)
            {
                var campoFormulario = (CampoFormulario)campo;
                Console.WriteLine($"{campoFormulario.NomeDoCampo}: {(campo.Validar() ? "Válido" : "Inválido")}");
            }
        }
    }
}
